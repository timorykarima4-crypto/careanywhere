<script>
let mode = 'short';
let history = [];
let responding = false;

function setMode(m) {
  mode = m;
  document.getElementById('btn-short').className = 'toggle-btn' + (m==='short'?' active':'');
  document.getElementById('btn-long').className = 'toggle-btn' + (m==='long'?' active':'');
}

function getPrompt() {
  const len = mode==='short'
    ? 'Keep EVERY response to 1-2 short sentences. One question at a time.'
    : 'Give 3-5 sentence responses. One question at a time.';
  return `You are CareAnywhere, a warm AI health navigator for rural and underserved communities worldwide.\n\nPERSONALITY: Warm, caring, plain language, natural and varied — never robotic.\n\nBANNED PHRASES — NEVER SAY THESE:\n- "we are here for you"\n- "we've got your back"\n- "you came to the right place"\n- "don't worry"\n- "we understand"\n- Any similar reassurance phrase\n\nVARIETY: Never start two responses the same way. Always vary your language.\n\nLENGTH: ${len}\n\nROLE: Help users understand symptoms, guide them to right care level, find nearby resources.\n\nSAFETY RULES:\n- NEVER diagnose\n- Send to 911/ER for: chest pain, difficulty breathing, severe bleeding, stroke, high fever in babies under 3 months\n- NEVER give specific medication doses\n- You are a navigator not a doctor\n\nGLOBAL AWARENESS: You serve rural America AND places like Afghanistan where medication access is extremely limited.`;
}

function addBubble(text, type) {
  const chat = document.getElementById('chat');
  const b = document.createElement('div');
  b.className = `bubble ${type}`;
  b.textContent = text;
  chat.appendChild(b);
  chat.scrollTop = chat.scrollHeight;
}

function showTyping() {
  removeTyping();
  const chat = document.getElementById('chat');
  const t = document.createElement('div');
  t.className = 'bubble typing'; t.id = 'typ';
  t.innerHTML = '<span class="dot"></span><span class="dot"></span><span class="dot"></span>';
  chat.appendChild(t);
  chat.scrollTop = chat.scrollHeight;
}

function removeTyping() { const t = document.getElementById('typ'); if(t) t.remove(); }

async function callAI(msg) {
  history.push({role:'user', content:msg});
  const r = await fetch('/api/chat', {
    method:'POST',
    headers:{'Content-Type':'application/json'},
    body: JSON.stringify({messages: history, system: getPrompt()})
  });
  if(!r.ok) throw new Error(r.status);
  const d = await r.json();
  const reply = d.content.filter(b=>b.type==='text').map(b=>b.text).join('');
  history.push({role:'assistant', content:reply});
  return reply;
}

async function sendMsg() {
  if(responding) return;
  const input = document.getElementById('msgInput');
  const text = input.value.trim();
  if(!text) return;
  input.value = '';
  responding = true;
  document.getElementById('sendBtn').disabled = true;
  addBubble(text, 'user');
  showTyping();
  try {
    const reply = await callAI(text);
    removeTyping();
    addBubble(reply, 'app');
  } catch(e) {
    removeTyping();
    addBubble("Having trouble connecting. Please try again.", 'app');
  }
  responding = false;
  document.getElementById('sendBtn').disabled = false;
  input.focus();
}

async function init() {
  showTyping();
  await new Promise(r=>setTimeout(r,900));
  removeTyping();
  const opening = "Hi, what's going on today? 😊 You came to the right place!";
  addBubble(opening, 'app');
  history.push({role:'assistant', content:opening});
  document.getElementById('msgInput').focus();
}

init();
</script>
